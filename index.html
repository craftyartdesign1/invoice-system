const SHEET_NAMES = { INVOICES: 'Invoices', INVENTORY: 'Inventory', CUSTOMERS: 'Customers', USERS: 'Users', SETTINGS: 'Settings' }; function doGet(e) { const action = e.parameter.action; if (action === 'login') return login(e.parameter); if (action === 'getAll') return createResponse(true, 'Retrieved', getAllData()); return HtmlService.createHtmlOutput('API Active'); } function doPost(e) { try { const data = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.getActiveSpreadsheet(); if (data.action === 'saveInvoice') { ss.getSheetByName(SHEET_NAMES.INVOICES).appendRow([ data.invoiceData.invoiceNumber, data.invoiceData.date, data.invoiceData.customer, "Crafty Art Design", "Admin", data.invoiceData.total, 0, data.invoiceData.total, JSON.stringify(data.invoiceData.items), new Date() ]); } else if (data.action === 'saveInventory') { ss.getSheetByName(SHEET_NAMES.INVENTORY).appendRow(data.data); } else if (data.action === 'saveCustomer') { ss.getSheetByName(SHEET_NAMES.CUSTOMERS).appendRow(data.data); } else if (data.action === 'deleteRow') { const sheet = ss.getSheetByName(data.sheet); const values = sheet.getDataRange().getValues(); // Searches the first column (ID/SKU/Name) to find the match to delete for (let i = 1; i < values.length; i++) { if (values[i][0].toString() === data.id.toString()) { sheet.deleteRow(i + 1); break; } } } return createResponse(true, 'Action Successful'); } catch (error) { return createResponse(false, 'Error: ' + error.toString()); } } function getAllData() { return { inventory: getSheetData(SHEET_NAMES.INVENTORY), customers: getSheetData(SHEET_NAMES.CUSTOMERS), settings: getSettingsData() }; } function getSheetData(name) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(name); if (!sheet) return []; const values = sheet.getDataRange().getValues(); if (values.length <= 1) return []; const headers = values[0].map(h => h.toString().toLowerCase().replace(/\s+/g, '')); return values.slice(1).map(row => { let obj = {}; headers.forEach((h, i) => obj[h] = row[i]); return obj; }); } function getSettingsData() { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAMES.SETTINGS); if (!sheet) return {}; const values = sheet.getDataRange().getValues(); let settings = {}; values.slice(1).forEach(row => { if(row[0]) settings[row[0]] = row[1]; }); return settings; } function login(p) { const users = getSheetData(SHEET_NAMES.USERS); const user = users.find(u => u.username === p.username && u.password === p.password); return createResponse(!!user, user ? 'Welcome' : 'Denied', user); } function createResponse(success, message, data = null) { return ContentService.createTextOutput(JSON.stringify({success, message, data})).setMimeType(ContentService.MimeType.JSON); }
