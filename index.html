function doGet(e) { const action = e.parameter.action; if (action === 'login') { return handleLogin(e); } // Add other actions (like fetching inventory) here later } function handleLogin(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName('Users'); // Matches your "Users" tab const data = sheet.getDataRange().getValues(); const username = e.parameter.username; const password = e.parameter.password; // Skip header row and look for user for (let i = 1; i < data.length; i++) { if (data[i][0] === username && data[i][1] === password) { const user = { username: data[i][0], role: data[i][2], display_name: data[i][3] }; return ContentService.createTextOutput(JSON.stringify({ success: true, user: user })) .setMimeType(ContentService.MimeType.JSON); } } return ContentService.createTextOutput(JSON.stringify({ success: false })) .setMimeType(ContentService.MimeType.JSON); }
